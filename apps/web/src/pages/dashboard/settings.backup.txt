        document.addEventListener("DOMContentLoaded", () => {
            // --- 1. Navigation Shell ---
            const pills = document.querySelectorAll(".nav-pill");
            const panels = document.querySelectorAll(".settings-panel");
            pills.forEach(p => p.addEventListener("click", () => {
                const group = p.getAttribute("data-group");
                pills.forEach(x => x.classList.toggle("active", x === p));
                panels.forEach(x => x.classList.toggle("active", x.id === `panel-${group}`));
                window.scrollTo({ top: 0, behavior: 'smooth' });
            }));

            // --- 2. Change Tracking ---
            const footer = document.querySelector(".settings-footer-sticky");
            const showSave = () => footer?.classList.add("show");
            document.querySelectorAll("input, select, textarea").forEach(i => {
                i.addEventListener("change", showSave);
                i.addEventListener("input", showSave);
            });

            // --- 3. Favicon Upload ---
            const favInput = document.getElementById('favicon-file-input');
            const favPreview = document.getElementById('favicon-preview-img');
            const favUrlVal = document.getElementById('favicon-url');
            favInput?.addEventListener('change', async (e) => {
                const file = e.target.files[0];
                if(!file) return;
                const fd = new FormData(); fd.append('file', file);
                try {
                   favPreview.style.opacity = '0.3';
